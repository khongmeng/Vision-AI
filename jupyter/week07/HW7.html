class TwoInputNN(nn.Module):
    def __init__(self):
        super().__init__()
        self.branch1 = nn.Sequential(
            nn.Linear(6, 16),
            nn.ReLU(),
            nn.Linear(16, 8),
            nn.ReLU()
        )
        self.branch2 = nn.Sequential(
            nn.Linear(7, 16),
            nn.ReLU(),
            nn.Linear(16, 8),
            nn.ReLU()
        )
        self.classifier = nn.Sequential(
            nn.Linear(16, 16),
            nn.ReLU(),
            nn.Linear(16, 8),
            nn.ReLU(),
            nn.Linear(8, 1)
        )
    def forward(self, x1, x2):
        z1 = self.branch1(x1)
        z2 = self.branch2(x2)
        z = torch.cat((z1, z2), dim=1)
        out = self.classifier(z)
        return out

def BCE(logits, targets, eps=1e-8):
    probs = torch.sigmoid(logits)
    probs = torch.clamp(probs, eps, 1 - eps)
    loss = -(targets * torch.log(probs) + (1 - targets) * torch.log(1 - probs))
    return loss.mean()

def train(model, loader, epochs=200, lr=0.001):
    optimizer = optim.SGD(model.parameters(), lr=lr)
    loss_history = []
    for epoch in range(epochs):
        model.train()
        total_loss = 0
        total_samples = 0
        for xb1, xb2, yb in loader:
            xb1 = xb1.to(device)
            xb2 = xb2.to(device)
            yb  = yb.to(device)

            optimizer.zero_grad()
            logits = model(xb1, xb2)
            loss = BCE(logits, yb)
            loss.backward()
            optimizer.step()
            batch_size = yb.size(0)
            total_loss += loss.item() * batch_size
            total_samples += batch_size

        epoch_loss = total_loss / total_samples
        loss_history.append(epoch_loss)
        
        if (epoch + 1) % 20 == 0 or epoch == 0:
            print(f"Epoch {epoch+1}/{epochs}, Loss: {epoch_loss:.6f}")
    return loss_history

def evaluate(model, loader, threshold=0.5):
    model.eval()

    all_preds = []
    all_targets = []

    with torch.no_grad():
        for xb1, xb2, yb in loader:
            xb1 = xb1.to(device)
            xb2 = xb2.to(device)

            logits = model(xb1, xb2)
            probs = torch.sigmoid(logits)
            preds = (probs >= threshold).float()

            all_preds.extend(preds.cpu().numpy().flatten())
            all_targets.extend(yb.cpu().numpy().flatten())

    all_preds = np.array(all_preds).astype(int)
    all_targets = np.array(all_targets).astype(int)

    cm = confusion_matrix(all_targets, all_preds)
    acc = (all_preds == all_targets).mean()

    return cm, acc, all_preds
